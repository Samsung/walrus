# Older CMake ignores the policy upper bound; CMake 4.x requires it.
CMAKE_MINIMUM_REQUIRED (VERSION 2.8.12...4.0 FATAL_ERROR)

PROJECT (WALRUS)

# CONFIGURATION
SET (CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Honor standard CMake configuration variables. Legacy inputs remain aliases.
IF (NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    IF (DEFINED WALRUS_MODE)
        IF (WALRUS_MODE STREQUAL "debug")
            SET (_walrus_build_type Debug)
        ELSEIF (WALRUS_MODE STREQUAL "release")
            SET (_walrus_build_type Release)
        ELSE()
            MESSAGE (FATAL_ERROR "WALRUS_MODE must be debug or release")
        ENDIF()
    ELSE()
        SET (_walrus_build_type Release)
    ENDIF()
    SET (CMAKE_BUILD_TYPE "${_walrus_build_type}" CACHE STRING "Build configuration" FORCE)
    SET_PROPERTY (CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release RelWithDebInfo MinSizeRel)
ENDIF()
STRING (TOLOWER "${CMAKE_BUILD_TYPE}" _walrus_build_type_lower)
IF (_walrus_build_type_lower STREQUAL "debug")
    SET (WALRUS_MODE debug)
ELSE()
    SET (WALRUS_MODE release)
ENDIF()

# Detect multilib targets without changing the caller's platform variables.
SET (_walrus_processor "${CMAKE_SYSTEM_PROCESSOR}")
IF (CMAKE_SIZEOF_VOID_P EQUAL 4 AND _walrus_processor MATCHES "^(x86_64|AMD64|x64)$")
    SET (_walrus_processor i386)
ENDIF()
IF (("${CMAKE_C_FLAGS} ${CMAKE_CXX_FLAGS} $ENV{CFLAGS} $ENV{CXXFLAGS} $ENV{LDFLAGS}" MATCHES "-m32")
    AND (_walrus_processor MATCHES "^(x86_64|AMD64|x64)$"))
    SET (_walrus_processor i386)
ENDIF()
IF (NOT _walrus_processor AND CMAKE_OSX_ARCHITECTURES)
    LIST (LENGTH CMAKE_OSX_ARCHITECTURES _walrus_arch_count)
    IF (_walrus_arch_count EQUAL 1)
        SET (_walrus_processor "${CMAKE_OSX_ARCHITECTURES}")
    ENDIF()
ENDIF()

IF (NOT DEFINED WALRUS_ARCH)
    IF (_walrus_processor STREQUAL "X86" OR _walrus_processor STREQUAL "x86"
        OR _walrus_processor STREQUAL "i386" OR _walrus_processor STREQUAL "i686")
        SET(WALRUS_ARCH "x86")
    ELSEIF (_walrus_processor STREQUAL "AMD64" OR _walrus_processor STREQUAL "x86_64"
            OR _walrus_processor STREQUAL "x64" OR _walrus_processor STREQUAL "EM64T")
        SET(WALRUS_ARCH "x64")
    ELSEIF (_walrus_processor STREQUAL "arm" OR _walrus_processor STREQUAL "arm32"
            OR _walrus_processor STREQUAL "armv7l")
        SET(WALRUS_ARCH "arm")
    ELSEIF (_walrus_processor STREQUAL "arm64" OR _walrus_processor STREQUAL "ARM64" OR _walrus_processor STREQUAL "aarch64" OR _walrus_processor STREQUAL "AARCH64")
        SET(WALRUS_ARCH "aarch64")
    ELSEIF (_walrus_processor STREQUAL "riscv64")
        SET(WALRUS_ARCH "riscv64")
    ENDIF()
ENDIF()

IF (NOT DEFINED WALRUS_HOST)
    IF (CMAKE_SYSTEM_NAME STREQUAL "Windows" OR CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
        SET(WALRUS_HOST "windows")
    ELSEIF (CMAKE_SYSTEM_NAME STREQUAL "Android")
        SET(WALRUS_HOST "android")
    ELSEIF (CMAKE_SYSTEM_NAME STREQUAL "Darwin")
        SET(WALRUS_HOST "darwin")
    ELSEIF (CMAKE_SYSTEM_NAME STREQUAL "Tizen")
        SET(WALRUS_HOST "tizen")
    ELSE()
        SET(WALRUS_HOST "linux")
    ENDIF()
ENDIF()

# Preserve parent-provided normal variables on old CMake too (CMP0077 only
# exists since 3.13), and keep all inputs visible as BOOL cache entries.
MACRO (WALRUS_OPTION _name _help _default)
    IF (DEFINED ${_name})
        SET (${_name} "${${_name}}" CACHE BOOL "${_help}")
    ENDIF()
    OPTION (${_name} "${_help}" "${_default}")
ENDMACRO()

# Project-specific booleans inherit standard defaults without modifying them.
SET (_walrus_shared_default OFF)
IF (DEFINED BUILD_SHARED_LIBS)
    SET (_walrus_shared_default "${BUILD_SHARED_LIBS}")
ENDIF()
SET (_walrus_shell_default ON)
IF (DEFINED ENABLE_SHELL)
    SET (_walrus_shell_default "${ENABLE_SHELL}")
ENDIF()
SET (_walrus_api_test_default OFF)
IF (DEFINED WALRUS_OUTPUT)
    IF (WALRUS_OUTPUT STREQUAL "shared_lib")
        SET (_walrus_shared_default ON)
        SET (_walrus_shell_default OFF)
    ELSEIF (WALRUS_OUTPUT STREQUAL "static_lib")
        SET (_walrus_shared_default OFF)
        SET (_walrus_shell_default OFF)
    ELSEIF (WALRUS_OUTPUT STREQUAL "api_test")
        SET (_walrus_shared_default OFF)
        SET (_walrus_shell_default OFF)
        SET (_walrus_api_test_default ON)
    ELSEIF (WALRUS_OUTPUT STREQUAL "shell")
        SET (_walrus_shell_default ON)
    ELSE()
        MESSAGE (FATAL_ERROR "Unknown WALRUS_OUTPUT: ${WALRUS_OUTPUT}")
    ENDIF()
ENDIF()
WALRUS_OPTION (WALRUS_BUILD_SHARED_LIBS "Build the Walrus library as a shared library" ${_walrus_shared_default})
WALRUS_OPTION (WALRUS_BUILD_GC_SHARED_LIBS "Build GCutil as a shared library" OFF)
WALRUS_OPTION (WALRUS_ENABLE_SHELL "Build the Walrus command-line shell" ${_walrus_shell_default})
WALRUS_OPTION (WALRUS_BUILD_API_TESTS "Build wasm-c-api examples" ${_walrus_api_test_default})
WALRUS_OPTION (WALRUS_WASI "Enable WASI support" ON)
WALRUS_OPTION (WALRUS_WASI_NN "Enable WASI neural-network support (requires CMake 3.14+)" OFF)
WALRUS_OPTION (WALRUS_GC "Enable WebAssembly garbage collection" ON)
WALRUS_OPTION (WALRUS_JIT "Enable the JIT compiler" ON)
WALRUS_OPTION (WALRUS_ASAN "Build with AddressSanitizer" OFF)
WALRUS_OPTION (WALRUS_SMALL_CONFIG "Optimize for binary size" OFF)
WALRUS_OPTION (WALRUS_DEBUG_INFO "Include debug information" OFF)
WALRUS_OPTION (WALRUS_VALGRIND "Enable Valgrind profiling" OFF)
WALRUS_OPTION (WALRUS_GOOGLE_PERF "Enable Google performance profiling" OFF)
WALRUS_OPTION (WALRUS_COVERAGE "Enable code coverage" OFF)
WALRUS_OPTION (WALRUS_JITPERF "Enable JIT perf support" OFF)
IF (WALRUS_WASI_NN AND NOT WALRUS_WASI)
    MESSAGE (FATAL_ERROR "WALRUS_WASI_NN requires WALRUS_WASI=ON")
ENDIF()
IF (WALRUS_WASI_NN AND CMAKE_VERSION VERSION_LESS 3.14)
    MESSAGE (FATAL_ERROR "WALRUS_WASI_NN requires CMake 3.14 or newer")
ENDIF()

MESSAGE(STATUS "Walrus Arch: " ${WALRUS_ARCH})
MESSAGE(STATUS "Walrus Host: " ${WALRUS_HOST})
MESSAGE(STATUS "Walrus Mode: " ${WALRUS_MODE})
MESSAGE(STATUS "Walrus Shared Library: " ${WALRUS_BUILD_SHARED_LIBS})
MESSAGE(STATUS "Walrus Shell: " ${WALRUS_ENABLE_SHELL})
MESSAGE(STATUS "Walrus API Tests: " ${WALRUS_BUILD_API_TESTS})
MESSAGE(STATUS "Walrus WASI Enabled: " ${WALRUS_WASI})
MESSAGE(STATUS "Walrus GC Enabled: " ${WALRUS_GC})
MESSAGE(STATUS "Walrus WASI-NN Enabled: " ${WALRUS_WASI_NN})
IF(WALRUS_WASI_NN)
    MESSAGE(STATUS "Walrus WASI-NN Backend: " ${WASI_NN_BACKEND})
ENDIF()

SET (WALRUS_TARGET walrus)

INCLUDE (ProcessorCount)
PROCESSORCOUNT (NPROCS)

INCLUDE (${CMAKE_CURRENT_SOURCE_DIR}/build/config.cmake)
INCLUDE (${CMAKE_CURRENT_SOURCE_DIR}/build/walrus.cmake)
