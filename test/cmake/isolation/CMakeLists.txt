cmake_minimum_required(VERSION 2.8.12...4.0)
project(WalrusIsolation C CXX)

get_filename_component(WALRUS_ISOLATION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
# Parent settings deliberately differ from Walrus defaults.
set(CMAKE_EXPORT_COMPILE_COMMANDS OFF CACHE BOOL "Parent preference" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE OFF)
set(CMAKE_CXX_STANDARD 17)
set_property(DIRECTORY APPEND PROPERTY COMPILE_OPTIONS -DPARENT_DIRECTORY_OPTION=1)
set_property(DIRECTORY APPEND PROPERTY COMPILE_DEFINITIONS PARENT_DIRECTORY_DEFINITION=1)
set_property(DIRECTORY APPEND PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -DPARENT_C_FLAG=1")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -DPARENT_CXX_FLAG=1")
set(WALRUS_ISOLATION_VARIABLES CMAKE_BUILD_TYPE CMAKE_C_FLAGS CMAKE_CXX_FLAGS
    CMAKE_EXE_LINKER_FLAGS CMAKE_SHARED_LINKER_FLAGS CMAKE_MODULE_PATH
    CMAKE_CXX_STANDARD CMAKE_CXX_STANDARD_REQUIRED CMAKE_CXX_EXTENSIONS
    CMAKE_POSITION_INDEPENDENT_CODE CMAKE_EXPORT_COMPILE_COMMANDS)
foreach(NAME ${WALRUS_ISOLATION_VARIABLES})
  set(WALRUS_ISOLATION_BEFORE_${NAME} "${${NAME}}")
endforeach()
set(WALRUS_ISOLATION_PROPERTIES COMPILE_OPTIONS COMPILE_DEFINITIONS
    INCLUDE_DIRECTORIES LINK_DIRECTORIES)
foreach(NAME ${WALRUS_ISOLATION_PROPERTIES})
  get_directory_property(WALRUS_ISOLATION_BEFORE_${NAME} ${NAME})
endforeach()
get_property(WALRUS_ISOLATION_CACHE_BEFORE GLOBAL PROPERTY CACHE_VARIABLES)

add_subdirectory("${WALRUS_ISOLATION_ROOT}" walrus)

foreach(NAME ${WALRUS_ISOLATION_VARIABLES})
  if (NOT "${${NAME}}" STREQUAL "${WALRUS_ISOLATION_BEFORE_${NAME}}")
    message(FATAL_ERROR "Walrus changed parent variable ${NAME}")
  endif ()
endforeach()
# A normal-variable shadow must not hide a modified shared cache entry.
get_property(WALRUS_ISOLATION_CACHE_BUILD_TYPE CACHE CMAKE_BUILD_TYPE PROPERTY VALUE)
if (NOT "${WALRUS_ISOLATION_CACHE_BUILD_TYPE}" STREQUAL "${WALRUS_ISOLATION_BEFORE_CMAKE_BUILD_TYPE}")
  message(FATAL_ERROR "Walrus changed the shared CMAKE_BUILD_TYPE cache")
endif ()
foreach(DIRECTORY "${WALRUS_ISOLATION_ROOT}"
    "${WALRUS_ISOLATION_ROOT}/third_party/wabt"
    "${WALRUS_ISOLATION_ROOT}/third_party/uvwasi"
    "${WALRUS_ISOLATION_ROOT}/third_party/GCutil")
  foreach(NAME ${WALRUS_ISOLATION_VARIABLES})
    if (NOT NAME MATCHES "^CMAKE_BUILD_TYPE$")
      get_directory_property(ACTUAL DIRECTORY "${DIRECTORY}" DEFINITION ${NAME})
      if (NOT "${ACTUAL}" STREQUAL "${WALRUS_ISOLATION_BEFORE_${NAME}}")
        message(FATAL_ERROR "Directory changed ${NAME} in ${DIRECTORY}: ${ACTUAL}")
      endif ()
    endif ()
  endforeach()
  foreach(NAME ${WALRUS_ISOLATION_PROPERTIES})
    get_directory_property(ACTUAL DIRECTORY "${DIRECTORY}" ${NAME})
    if (NOT "${ACTUAL}" STREQUAL "${WALRUS_ISOLATION_BEFORE_${NAME}}")
      message(FATAL_ERROR "Directory-scoped ${NAME} in ${DIRECTORY}: ${ACTUAL}")
    endif ()
  endforeach()
endforeach()
get_property(WALRUS_ISOLATION_CACHE_AFTER GLOBAL PROPERTY CACHE_VARIABLES)
foreach(NAME ${WALRUS_ISOLATION_CACHE_AFTER})
  list(FIND WALRUS_ISOLATION_CACHE_BEFORE "${NAME}" EXISTED)
  if (EXISTED EQUAL -1 AND NAME MATCHES "^(USE_(ASAN|MSAN|LSAN|UBSAN)|ASAN|WERROR|WITH_EXCEPTIONS|WITH_WASI|HAVE_(ALLOCA_H|UNISTD_H|SNPRINTF|STRCASECMP|WIN32_VT100|SYS_TYPES_H|STDINT_H|STDDEF_H|SSIZE_T)|SSIZE_T|SIZEOF_SIZE_T|HAS_UBSAN_RECOVER_(BARE|ALL))$")
    message(FATAL_ERROR "Unprefixed dependency cache variable: ${NAME}")
  endif ()
endforeach()

add_executable(walrus_isolation_c sentinel.c)
add_executable(walrus_isolation_cxx sentinel.cpp)
foreach(TARGET walrus_isolation_c walrus_isolation_cxx)
  get_target_property(PIC ${TARGET} POSITION_INDEPENDENT_CODE)
  if (PIC)
    message(FATAL_ERROR "Walrus enabled PIC on an unrelated target")
  endif ()
endforeach()
add_custom_target(walrus-isolation DEPENDS walrus_isolation_c walrus_isolation_cxx)
message(STATUS "Walrus parent, directory and cache isolation checks passed")
